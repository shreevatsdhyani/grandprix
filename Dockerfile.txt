FROM python:3.12-slim

  # Install system dependencies (ffmpeg is critical for audio processing)
RUN apt-get update && apt-get install -y --no-install-recommends \
    ffmpeg \
    git \
    curl \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /app

  # Copy and install Python dependencies
COPY backend/requirements.txt ./requirements.txt
RUN pip install --no-cache-dir -r requirements.txt \
    --extra-index-url https://download.pytorch.org/whl/cpu

  # Copy backend code and trained model files
COPY backend/ ./backend/
COPY data/labels/ ./data/labels/

  # Pre-download HF models at build time (makes container offline-ready)
RUN python backend/scripts/warm_models.py

  # HuggingFace Spaces expects port 7860
EXPOSE 7860
ENV PORT=7860

  # Start the FastAPI server
CMD ["uvicorn", "backend.app.main:app", "--host", "0.0.0.0", "--port", "7860"]
